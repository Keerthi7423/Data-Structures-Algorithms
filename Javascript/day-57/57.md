function findMin(nums){

    let min=nums[0];

    for(let i=1;i<nums.length;i++){

        if(nums[i]<min){

            min=nums[i];

        }

    }

    return min;

}

16. OPTIMIZED CODE
==================================================

function findMin(nums){

    let left=0;

    let right=
    nums.length-1;

    while(left<right){

        let mid=
        Math.floor(
            (left+right)/2
        );

        if(
            nums[mid] >
            nums[right]
        ){

            left=mid+1;

        }
        else{

            right=mid;

        }

    }

    return nums[left];

}
