5. BRUTE FORCE CODE

function maxSubArray(nums){

    let maxSum = -Infinity;

    for(let i=0;i<nums.length;i++){

        let currentSum = 0;

        for(let j=i;j<nums.length;j++){

            currentSum += nums[j];

            maxSum =
            Math.max(maxSum,currentSum);

        }

    }

    return maxSum;

}

function maxSubArray(nums){

    let currentSum = 0;

    let maxSum = nums[0];

    for(let num of nums){

        currentSum += num;

        maxSum =
        Math.max(maxSum,currentSum);

        if(currentSum<0){

            currentSum=0;

        }

    }

    return maxSum;

}
