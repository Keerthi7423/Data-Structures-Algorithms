function maxArea(height){

    let maxWater=0;

    for(let i=0;i<height.length;i++){

        for(let j=i+1;j<height.length;j++){

            let width=j-i;

            let h=Math.min(
                height[i],
                height[j]
            );

            let area=width*h;

            maxWater=
            Math.max(
                maxWater,
                area
            );

        }

    }

    return maxWater;

}

function maxArea(height){

    let left=0;

    let right=
    height.length-1;

    let maxWater=0;

    while(left<right){

        let width=
        right-left;

        let h=
        Math.min(
            height[left],
            height[right]
        );

        let area=
        width*h;

        maxWater=
        Math.max(
            maxWater,
            area
        );

        if(
            height[left]
            <
            height[right]
        ){

            left++;

        }
        else{

            right--;

        }

    }

    return maxWater;

}
