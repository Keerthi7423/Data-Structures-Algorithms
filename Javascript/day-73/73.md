function moveZeroes(nums){

    let result=[];

    let zeroCount=0;

    for(let num of nums){

        if(num!==0){

            result.push(num);

        }

        else{

            zeroCount++;

        }

    }

    while(zeroCount>0){

        result.push(0);

        zeroCount--;

    }

    return result;

}

function moveZeroes(nums){

    let write=0;

    for(let read=0;read<nums.length;read++){

        if(nums[read]!==0){

            nums[write]=nums[read];

            write++;

        }

    }

    while(write<nums.length){

        nums[write]=0;

        write++;

    }

    return nums;

}
