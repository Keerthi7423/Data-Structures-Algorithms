function findPeakElement(nums){

    for(let i=0;i<nums.length;i++){

        let left =
        i===0
        ? -Infinity
        : nums[i-1];

        let right =
        i===nums.length-1
        ? -Infinity
        : nums[i+1];

        if(
            nums[i] > left &&
            nums[i] > right
        ){

            return i;

        }

    }

}

15. OPTIMIZED CODE
==================================================

function findPeakElement(nums){

    let left=0;

    let right=
    nums.length-1;

    while(left<right){

        let mid=
        Math.floor(
            (left+right)/2
        );

        if(
            nums[mid] <
            nums[mid+1]
        ){

            left=mid+1;

        }
        else{

            right=mid;

        }

    }

    return left;

}
