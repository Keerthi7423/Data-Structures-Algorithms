function isPerfectSquare(num){

    for(let i=1;i<=num;i++){

        if(i*i===num){

            return true;

        }

        if(i*i>num){

            return false;

        }

    }

function isPerfectSquare(num){

    let left=1;

    let right=num;

    while(left<=right){

        let mid=
        Math.floor(
            (left+right)/2
        );

        let square=
        mid*mid;

        if(square===num){

            return true;

        }

        else if(square<num){

            left=mid+1;

        }

        else{

            right=mid-1;

        }

    }

    return false;

}
