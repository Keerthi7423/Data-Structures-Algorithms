function findMaxAverage(nums, k){

    let maxAverage = -Infinity;

    for(let i = 0; i <= nums.length - k; i++){

        let sum = 0;

        for(let j = i; j < i + k; j++){

            sum += nums[j];

        }

        maxAverage = Math.max(maxAverage, sum / k);

    }

    return maxAverage;

}

function findMaxAverage(nums, k){

    let sum = 0;

    for(let i = 0; i < k; i++){

        sum += nums[i];

    }

    let maxSum = sum;

    for(let right = k; right < nums.length; right++){

        sum += nums[right];

        sum -= nums[right - k];

        maxSum = Math.max(maxSum, sum);

    }

    return maxSum / k;

}
