function twoSum(numbers,target){

    for(let i=0;i<numbers.length;i++){

        for(let j=i+1;j<numbers.length;j++){

            if(

                ===target

function twoSum(numbers,target){

    let left=0;

    let right=
    numbers.length-1;

    while(left<right){

        let sum=
        numbers[left]
        +
        numbers[right];

        if(sum===target){

            return[
                left+1,
                right+1
            ];

        }

        if(sum<target){

            left++;

        }

        else{

            right--;

        }

    }

}
