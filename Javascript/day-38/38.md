==================================================
0. PROBLEM IN SIMPLE WORDS
==================================================

For every position:

Multiply all other numbers
except current number

--------------------------------------------------

Example:

nums = [1,2,3,4]

For index 0:

2 × 3 × 4 = 24

For index 1:

1 × 3 × 4 = 12

For index 2:

1 × 2 × 4 = 8

For index 3:

1 × 2 × 3 = 6

Final Answer:

[24,12,8,6]

5. BRUTE FORCE CODE
==================================================

function productExceptSelf(nums) {
    let result = [];

    for (let i = 0; i < nums.length; i++) {
        let product = 1;

        for (let j = 0; j < nums.length; j++) {
            if (i !== j) {
                product *= nums[j];
            }
        }

        result.push(product);
    }

    return result;
}

10. NEW ALGORITHM → PREFIX + SUFFIX
==================================================

This is actual interview solution

--------------------------------------------------

PREFIX:

Product of left side

Example:

nums = [1,2,3,4]

Prefix:

[1,1,2,6]

--------------------------------------------------

Why?

Before 1 → nothing → 1

Before 2 → only 1

Before 3 → 1×2

Before 4 → 1×2×3

--------------------------------------------------

SUFFIX:

Product of right side

Suffix:

[24,12,4,1]

==================================================

Multiply prefix × suffix

[24,12,8,6]

==================================================
13. OPTIMIZED CODE
==================================================

function productExceptSelf(nums) {
    let result = new Array(nums.length).fill(1);

    let prefix = 1;

    for (let i = 0; i < nums.length; i++) {
        result[i] = prefix;
        prefix *= nums[i];
    }

    let suffix = 1;

    for (let i = nums.length - 1; i >= 0; i--) {
        result[i] *= suffix;
        suffix *= nums[i];
    }

    return result;
}
