function reverseWords(s){

    let words=[];

    let i=0;

    while(i<s.length){

        while(i<s.length && s[i]===" "){

            i++;

        }

        let word="";

        while(i<s.length && s[i]!==" "){

            word+=s[i];

            i++;

        }

        if(word.length>0){

            words.push(word);

        }

    }

    words.reverse();

    return words.join(" ");

}

// Test Cases
