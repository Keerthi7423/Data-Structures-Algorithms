4. BRUTE FORCE APPROACH
==================================================

Logic:

Visual:

==================================================
5. BRUTE FORCE CODE
==================================================

function containsDuplicate(nums) {
    for(let i = 0; i < nums.length; i++) {
        for(let j = i + 1; j < nums.length; j++) {
            if(nums[i] === nums[j]) {
                return true;
            }
        }
    }

    return false;
}

8. BETTER APPROACH → SORTING
==================================================

Sort array first

Duplicates become adjacent

Example:

[4,1,3,1]

↓

[1,1,3,4]


==================================================
9. SORTING CODE
==================================================

function containsDuplicate(nums) {
    nums.sort((a,b) => a-b);

    for(let i = 1; i < nums.length; i++) {
        if(nums[i] === nums[i-1]) {
            return true;
        }
    }

    return false;
}

11. HASHSET CONCEPT
==================================================

Set helps quickly check whether
an element has already been seen.

Average lookup:
O(1)

Example:

let set = new Set()

set.add(1)
set.add(2)
set.add(1)

Final:

{1,2}


==================================================
12. HASHSET OPERATIONS
==================================================

Add:
set.add(value)

Check:
set.has(value)

Delete:
set.delete(value)


==================================================
13. WHY HASHSET HERE?
==================================================

Need fast duplicate lookup

Instead of comparing every pair,
we check instantly whether value exists.

15. BEST OPTIMIZED SOLUTION
==================================================

function containsDuplicate(nums) {
    let seen = new Set();

    for(let num of nums) {
        if(seen.has(num)) {
            return true;
        }

        seen.add(num);
    }

    return false;
}
