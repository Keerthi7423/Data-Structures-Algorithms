5. BRUTE FORCE CODE

function searchInsert(nums,target){

    for(let i=0;i<nums.length;i++){

        if(nums[i]>=target){

            return i;

        }

    }

    return nums.length;

}

14. WHY BINARY SEARCH WORKS HERE
==================================================

Array is sorted.

16. OPTIMIZED CODE
==================================================

function searchInsert(nums,target){

    let left=0;

    let right=
    nums.length-1;

    while(left<=right){

        let mid=
        Math.floor(
            (left+right)/2
        );

        if(nums[mid]===target){

            return mid;

        }

        else if(
            nums[mid]<target
        ){

            left=mid+1;

        }

        else{

            right=mid-1;

        }

    }

    return left;

}
