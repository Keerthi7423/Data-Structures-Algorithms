function reverseString(s){

    let result=[];

    for(let i=s.length-1;i>=0;i--){

        result.push(s[i]);

    }

    return result;

}

14. OPTIMIZED CODE
==================================================

function reverseString(s){

    let left=0;

    let right=s.length-1;

    while(left<right){

        let temp=s[left];

        s[left]=s[right];

        s[right]=temp;

        left++;

        right--;

    }

    return s;

}
